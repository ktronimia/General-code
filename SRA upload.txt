https://www.ibm.com/products/aspera/downloads?utm

2. unzip
3. go to the folder and run following code

chmod +x ibm-aspera-connect_4.2.18.918-HEAD_linux_x86_64.sh
./ibm-aspera-connect_4.2.18.918-HEAD_linux_x86_64.sh
~/.aspera/connect/bin/ascp --version
echo 'export PATH=$PATH:$HOME/.aspera/connect/bin' >> ~/.bashrc
source ~/.bashrc
